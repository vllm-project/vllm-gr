# must be built for that image's CANN version. vllm_gr._env_check pins the
# validated vLLM-Ascend/Triton-Ascend pair at startup.

# exactly (omitting the pin lets pip silently resolve numpy 2.x instead), so
# the Ascend runtime stays on numpy 1.26.4.
numpy==1.26.4
